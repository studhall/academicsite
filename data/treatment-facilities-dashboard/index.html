withTrendSegments = (rows, seriesAccessor = (d) => d.label) => {
  const output = []
  for (const [series, values] of d3.groups(rows, seriesAccessor)) {
    let segment = 0
    let previousYear = null
    for (const row of values.slice().sort((a, b) => a.survey_year - b.survey_year)) {
      if (
        previousYear !== null &&
        (row.survey_year - previousYear > 1 || (previousYear <= 2020 && row.survey_year >= 2021))
      ) segment += 1
      output.push({...row, series_segment: String(series) + "|" + String(segment)})
      previousYear = row.survey_year
    }
  }
  return output
}

wrapCanvasText = (context, text, x, y, maxWidth, lineHeight, maxLines = 3) => {
  const words = text.split(/\s+/)
  const lines = []
  let line = ""
  for (const word of words) {
    const test = line ? `${line} ${word}` : word
    if (context.measureText(test).width > maxWidth && line) {
      lines.push(line)
      line = word
    } else {
      line = test
    }
  }
  if (line) lines.push(line)
  lines.slice(0, maxLines).forEach((value, index) => context.fillText(value, x, y + index * lineHeight))
}

About these data

The analytical year is the survey year; the directory year is retained in labels and downloads. Counts represent directory listings rather than linked cross-year organizations. Lines intentionally break between the 2020 N-SSATS and 2021 N-SUMHSS survey years because SAMHSA advises against direct statistical comparison across the redesign.

The ownership chart uses SAMHSA public-use survey files, restricted to substance-use or combined-focus facilities in the 50 states and DC. Directory listings and public-use survey records have different inclusion rules, so the survey-file ownership series is not used for address-level filters or directory totals.

Multiple selections within ownership, center type, setting, and payment use OR logic. Filter groups combine with AND logic. Multiple services use the All/Any selection shown above.

Limitations and downloads

Quality assurance is ongoing, and these parser outputs may change during manual review. A gap means that a characteristic was not asked or cannot be harmonized for that year. County assignments include confidence and method fields. From 2000 onward, current assignments are ZIP-crosswalk approximations rather than street-geocoded locations.
